    71→type Direction = "higher" | "lower" | "neutral";
    72→
    73→/** Clés numériques de OverallMetrics (exclut bottleneckNodeId). */
    74→type NumericMetricKey = {
    75→  [K in keyof OverallMetrics]: OverallMetrics[K] extends number ? K : never;
    76→}[keyof OverallMetrics];
    77→
    78→interface MetricDef {
    79→  key: NumericMetricKey;
    80→  label: string;
    81→  format: (n: number) => string;
    82→  direction: Direction;
    83→  deltaFormat: (n: number) => string;
    84→}
    85→
    86→const sign = (n: number): string => (n > 0 ? "+" : n < 0 ? "−" : "±");
    87→
    88→const signed =
    89→  (fmt: (n: number) => string) =>
    90→  (n: number): string =>
    91→    `${sign(n)}${fmt(Math.abs(n))}`;
    92→
    93→const points = (n: number): string =>
    94→  `${sign(n)}${(Math.abs(n) * 100).toFixed(1).replace(".", ",")} pt`;
    95→
    96→const METRICS: MetricDef[] = [
    97→  { key: "servedRps", label: "Débit traité", format: formatRps, direction: "higher", deltaFormat: signed(formatRps) },
    98→  { key: "inputRps", label: "Débit entrant", format: formatRps, direction: "higher", deltaFormat: signed(formatRps) },
    99→  { key: "latencyAvgMs", label: "Latence moyenne", format: formatMs, direction: "lower", deltaFormat: signed(formatMs) },
   100→  { key: "p50Ms", label: "Latence p50", format: formatMs, direction: "lower", deltaFormat: signed(formatMs) },
   101→  { key: "p95Ms", label: "Latence p95", format: formatMs, direction: "lower", deltaFormat: signed(formatMs) },
   102→  { key: "p99Ms", label: "Latence p99", format: formatMs, direction: "lower", deltaFormat: signed(formatMs) },
   103→  { key: "errorRate", label: "Taux d'erreur", format: formatPercent, direction: "lower", deltaFormat: points },
   104→  { key: "rejectedRequests", label: "Requêtes rejetées", format: formatRequests, direction: "lower", deltaFormat: signed(formatRequests) },
   105→  { key: "availability", label: "Disponibilité", format: formatPercent, direction: "higher", deltaFormat: points },
   106→  { key: "costPerMonthUsd", label: "Coût estimé", format: formatCost, direction: "lower", deltaFormat: signed(formatCost) },
   107→  { key: "totalRequests", label: "Requêtes totales", format: formatRequests, direction: "neutral", deltaFormat: signed(formatRequests) },
   108→];
   109→
   110→function toneOf(def: MetricDef, a: number, b: number): DeltaTone {
   111→  if (def.direction === "neutral") return "neutral";
   112→  const diff = b - a;
   113→  if (Math.abs(diff) < 1e-9) return "neutral";
   114→  const better = def.direction === "higher" ? diff > 0 : diff < 0;
   115→  return better ? "better" : "worse";
   116→}

   122→interface SummaryAxis {

   126→  unit: "pct" | "points";

   129→const SUMMARY_AXES: SummaryAxis[] = [
   130→  { key: "latencyAvgMs", label: "la latence moyenne", direction: "lower", unit: "pct" },
   131→  { key: "p95Ms", label: "la latence p95", direction: "lower", unit: "pct" },
   132→  { key: "errorRate", label: "le taux d'erreur", direction: "lower", unit: "points" },
   133→  { key: "rejectedRequests", label: "les requêtes rejetées", direction: "lower", unit: "pct" },
   134→  { key: "servedRps", label: "le débit traité", direction: "higher", unit: "pct" },
   135→  { key: "availability", label: "la disponibilité", direction: "higher", unit: "points" },
   136→  { key: "costPerMonthUsd", label: "le coût estimé", direction: "lower", unit: "pct" },

   138→
   139→function joinFr(items: string[]): string {
   140→  if (items.length <= 1) return items.join("");
   141→  return `${items.slice(0, -1).join(", ")} et ${items[items.length - 1]}`;
   142→}
   143→
   144→function buildTradeoffSummary(a: SimulationResult, b: SimulationResult): string[] {
   145→  const improved: string[] = [];
   146→  const degraded: string[] = [];
   147→
   148→  for (const axis of SUMMARY_AXES) {
   149→    const va = a.overall[axis.key];
   150→    const vb = b.overall[axis.key];
   151→    const diff = vb - va;
   152→    if (Math.abs(diff) < 1e-12) continue;
   153→    const pct = va !== 0 ? diff / Math.abs(va) : null;
   154→    if (axis.unit === "pct") {
   155→      if (pct === null ? Math.abs(diff) < 1e-9 : Math.abs(pct) < 0.03) continue;
   156→    } else if (Math.abs(diff) < 0.001) continue;
   157→
   158→    const change =
   159→      axis.unit === "pct"
   160→        ? pct !== null
   161→          ? `${Math.round(Math.abs(pct) * 100)} %`
   162→          : formatRequests(Math.abs(diff))
   163→        : `${(Math.abs(diff) * 100).toFixed(1).replace(".", ",")} points`;
   164→    const item = `${axis.label} de ${change}`;
   165→    const better = axis.direction === "higher" ? diff > 0 : diff < 0;
   166→    (better ? improved : degraded).push(item);
   167→  }
   168→
   169→  const sentences: string[] = [];
   170→  if (improved.length > 0 && degraded.length > 0) {
   171→    sentences.push(
   172→      `La version B améliore ${joinFr(improved)} mais dégrade ${joinFr(degraded)}.`,
   173→    );
   174→  } else if (improved.length > 0) {
   175→    sentences.push(
   176→      `La version B améliore ${joinFr(improved)} sans dégradation notable des autres indicateurs.`,
   177→    );
   178→  } else if (degraded.length > 0) {
   179→    sentences.push(
   180→      `La version B dégrade ${joinFr(degraded)} sans gain mesurable sur les autres indicateurs.`,
   181→    );
   182→  } else {
   183→    sentences.push(
   184→      "Les deux versions affichent des écarts non significatifs (< 3 %) sur l'ensemble des indicateurs suivis.",
   185→    );
   186→  }
   187→
   188→  const p95Better = b.overall.p95Ms < a.overall.p95Ms;
   189→  const costBetter = b.overall.costPerMonthUsd < a.overall.costPerMonthUsd;
   190→  if (p95Better && costBetter) {
   191→    sentences.push(
   192→      "Verdict : la version B domine à la fois en latence et en coût — elle peut remplacer la version A.",
   193→    );
   194→  } else if (!p95Better && !costBetter) {
   195→    sentences.push(
   196→      "Verdict : la version A reste préférable sur la latence et le coût.",
   197→    );
   198→  } else if (p95Better) {
   199→    sentences.push(
   200→      "Verdict : le gain de latence se paie par un coût supérieur — pertinente si la priorité est la performance.",
   201→    );
   202→  } else {
   203→    sentences.push(
   204→      "Verdict : l'économie réalisée se paie en latence — pertinente si la priorité est le budget.",
   205→    );
   206→  }
   207→  return sentences;
   208→}

   214→function CompareSelector() {
   215→  const simulations = useAppStore((s) => s.simulations);
   216→  const simulationsLoading = useAppStore((s) => s.simulationsLoading);
   217→  const simulationsError = useAppStore((s) => s.simulationsError);
   218→  const refreshSimulations = useAppStore((s) => s.refreshSimulations);
   219→  const openCompare = useAppStore((s) => s.openCompare);
   220→  const setView = useAppStore((s) => s.setView);
   221→  const goBack = useAppStore((s) => s.goBack);
   222→  const viewHistory = useAppStore((s) => s.viewHistory);
   223→
   224→  const [idA, setIdA] = useState<string>("");
   225→  const [idB, setIdB] = useState<string>("");
   226→  const [loading, setLoading] = useState(false);
   227→
   228→  // Garde !loading (lecture fraîche via getState) : évite de relancer un
   229→  // chargement déjà en cours — le shell précharge au montage.
   230→  useEffect(() => {
   231→    const s = useAppStore.getState();
   232→    if (s.simulations.length === 0 && !s.simulationsLoading) {
   233→      void s.refreshSimulations();
   234→    }
   235→  }, [simulations.length, refreshSimulations]);
   236→
   237→  const selectedA = simulations.find((s) => s.id === idA);
   238→  const selectedB = simulations.find((s) => s.id === idB);
   239→
   240→  const handleCompare = async () => {
   241→    if (!idA || !idB || loading) return;
   242→    setLoading(true);
   243→    try {
   244→      const [resultA, resultB] = await Promise.all([
   245→        storage.simulations.get(idA),
   246→        storage.simulations.get(idB),
   247→      ]);
   248→      if (!resultA || !resultB) {
   249→        toast.error("Simulation introuvable", {
   250→          description: "Elle a peut-être été supprimée — actualisez la liste.",
   251→        });
   252→        return;
   253→      }
   254→      openCompare(resultA, resultB);
   255→    } catch (error) {
   256→      toast.error("Comparaison impossible", {
   257→        description: error instanceof Error ? error.message : "Erreur inconnue.",
   258→      });
   259→    } finally {
   260→      setLoading(false);
   261→    }
   262→  };
   263→
   264→  const renderColumn = (tag: "A" | "B") => {
   265→    const value = tag === "A" ? idA : idB;
   266→    const other = tag === "A" ? idB : idA;
   267→    const setValue = tag === "A" ? setIdA : setIdB;
   268→    const selected = tag === "A" ? selectedA : selectedB;
   269→    return (
   270→      <div className="space-y-2">
   271→        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   272→          Version {tag}
   273→        </p>
   274→        <Select value={value} onValueChange={setValue}>
   275→          <SelectTrigger className="w-full">
   276→            <SelectValue placeholder="Choisir une simulation archivée…" />
   277→          </SelectTrigger>
   278→          <SelectContent>
   279→            {simulations.map((sim) => (
   280→              <SelectItem key={sim.id} value={sim.id} disabled={sim.id === other}>
   281→                {sim.architectureName} · seed {sim.seed} · {formatDateTime(sim.createdAt)}
   282→              </SelectItem>
   283→            ))}
   284→          </SelectContent>
   285→        </Select>
   286→        {selected ? (
   287→          <p className="text-xs text-muted-foreground">
   288→            {formatRps(selected.overall.servedRps)} · p95 {formatMs(selected.overall.p95Ms)} ·{" "}
   289→            {formatPercent(selected.overall.errorRate)} d&apos;erreurs ·{" "}
   290→            {formatCost(selected.overall.costPerMonthUsd)}
   291→          </p>
   292→        ) : null}
   293→      </div>
   294→    );
   295→  };
   296→
   297→  return (
   298→    <div className="mx-auto w-full max-w-[1400px] p-6 pb-16">
   299→      <header className="flex flex-col gap-4 lg:flex-row lg:items-start lg:justify-between">
   300→        <div>
   301→          <h1 className="text-2xl font-semibold tracking-tight">Comparaison d&apos;architectures</h1>
   302→          <p className="mt-1 text-sm text-muted-foreground">
   303→            Sélectionnez deux simulations archivées pour comparer leurs métriques côte à côte.
   304→          </p>
   305→        </div>
   306→        <div className="flex flex-wrap items-center gap-2">
   307→          <Button
   308→            variant="outline"
   309→            size="sm"
   310→            onClick={() => void refreshSimulations()}
   311→            disabled={simulationsLoading}
   312→          >
   313→            <RefreshCw className={simulationsLoading ? "size-4 animate-spin" : "size-4"} />
   314→            Actualiser
   315→          </Button>
   316→          <Button
   317→            variant="ghost"
   318→            size="sm"
   319→            onClick={() => (viewHistory.length > 0 ? goBack() : setView("history"))}
   320→          >
   321→            <ArrowLeft className="size-4" />
   322→            Retour
   323→          </Button>
   324→        </div>
   325→      </header>
   326→
   327→      {simulationsError ? (
   328→        <div className="mt-16 flex justify-center">
   329→          <div className="w-full max-w-md">
   330→            <Alert variant="destructive" role="alert">
   331→              <CloudOff aria-hidden />
   332→              <AlertTitle>Serveur local injoignable</AlertTitle>
   333→              <AlertDescription>
   334→                Les simulations n&apos;ont pas pu être chargées — vos archives ne sont
   335→                pas perdues. Réessayez plus tard.
   336→              </AlertDescription>
   337→            </Alert>
   338→          </div>
   339→        </div>
   340→      ) : simulationsLoading && simulations.length === 0 ? (
   341→        <div className="mt-10 grid gap-6 md:grid-cols-2">
   342→          <Skeleton className="h-20 w-full" />
   343→          <Skeleton className="h-20 w-full" />
   344→        </div>
   345→      ) : simulations.length === 0 ? (
   346→        <div className="mt-16 flex flex-col items-center justify-center gap-5 text-center">
   347→          <span className="flex size-16 items-center justify-center rounded-2xl border bg-card shadow-sm">
   348→            <Inbox className="size-8 text-muted-foreground" />
   349→          </span>
   350→          <div className="space-y-1.5">
   351→            <h2 className="text-lg font-semibold">Aucune simulation archivée</h2>
   352→            <p className="mx-auto max-w-md text-sm text-muted-foreground">
   353→              Lancez au moins deux simulations depuis l&apos;éditeur pour pouvoir les comparer.
   354→            </p>
   355→          </div>
   356→          <Button onClick={() => setView("editor")}>Ouvrir l&apos;éditeur</Button>
   357→        </div>
   358→      ) : (
   359→        <>
   360→          <div className="mt-10 grid gap-6 md:grid-cols-2">
   361→            {renderColumn("A")}
   362→            {renderColumn("B")}
   363→          </div>
   364→          <div className="mt-8 flex flex-wrap items-center gap-3">
   365→            <Button onClick={handleCompare} disabled={!idA || !idB || loading}>
   366→              {loading ? (
   367→                <RefreshCw className="size-4 animate-spin" />
   368→              ) : (
   369→                <GitCompare className="size-4" />
   370→              )}
   371→              Comparer
   372→            </Button>
   373→            <p className="text-xs text-muted-foreground">
   374→              {simulations.length} simulation{simulations.length > 1 ? "s" : ""} archivée
   375→              {simulations.length > 1 ? "s" : ""} disponible
   376→              {simulations.length > 1 ? "s" : ""}.
   377→            </p>
   378→          </div>
   379→        </>
   380→      )}
   381→    </div>
   382→  );
   383→}

   502→                <TableHead className="max-w-[200px] truncate text-right" title={b.architectureName}>
   503→                  {nameB}
   504→                </TableHead>
   505→                <TableHead className="text-right">Δ (B − A)</TableHead>
   506→              </TableRow>
   507→            </TableHeader>
   508→            <TableBody>
   509→              {METRICS.map((def) => {
   510→                const va = a.overall[def.key];
   511→                const vb = b.overall[def.key];
   512→                const diff = vb - va;
   513→                const pct = va !== 0 ? diff / Math.abs(va) : null;
   514→                const relative =
   515→                  pct !== null && Math.abs(pct) >= 0.01
   516→                    ? `${pct > 0 ? "+" : "−"}${Math.round(Math.abs(pct) * 100)} %`
   517→                    : undefined;
   518→                return (
   519→                  <MetricRow
   520→                    key={def.key}
   521→                    label={def.label}
   522→                    valueA={def.format(va)}
   523→                    valueB={def.format(vb)}
   524→                    delta={def.deltaFormat(diff)}
   525→                    relative={relative}
   526→                    tone={toneOf(def, va, vb)}
   527→                  />
   528→                );
   529→              })}
   530→            </TableBody>
   531→          </Table>
   532→        </div>
   533→      </section>

   574→          <Card className="gap-3 py-4">

   576→              <CardTitle className="text-sm font-semibold">Volumétrie (requêtes)</CardTitle>
