   109→   109→function ScenarioNumberInput({
   110→   110→  id,
   111→   111→  value,
   112→   112→  min,
   113→   113→  max,
   114→   114→  step,
   115→   115→  onCommit,
   116→   116→  ariaLabel,
   117→   117→}: {
   118→   118→  id: string;
   119→   119→  value: number;
   120→   120→  min?: number;
   121→   121→  max?: number;
   122→   122→  step?: number;
   123→   123→  onCommit: (value: number) => void;
   124→   124→  ariaLabel: string;
   125→   125→}) {
   126→   126→  const [draft, setDraft] = useState(String(value));
   127→   127→  useEffect(() => {
   128→   128→    setDraft(String(value));
   129→   129→  }, [value]);
   130→   130→  const commit = (raw: string) => {
   131→   131→    const parsed = Number(raw.replace(",", "."));
   132→   132→    if (!Number.isFinite(parsed)) {
   133→   133→      setDraft(String(value));
   134→   134→      return;
   135→   135→    }
   136→   136→    let next = parsed;
   137→   137→    if (min !== undefined) next = Math.max(min, next);
   138→   138→    if (max !== undefined) next = Math.min(max, next);
   139→   139→    onCommit(next);
   140→   140→  };
   141→   141→  return (
   142→   142→    <Input
   143→   143→      id={id}
   144→   144→      type="number"
   145→   145→      inputMode="decimal"
   146→   146→      value={draft}
   147→   147→      min={min}
   148→   148→      max={max}
   149→   149→      step={step}
   150→   150→      aria-label={ariaLabel}
   151→   151→      onChange={(event) => setDraft(event.target.value)}
   152→   152→      onBlur={() => commit(draft)}
   153→   153→      onKeyDown={(event) => {
   154→   154→        if (event.key === "Enter") commit(draft);
   155→   155→      }}
   156→   156→      className="h-8 tabular-nums"
   157→   157→    />
   158→   158→  );
   159→   159→}
