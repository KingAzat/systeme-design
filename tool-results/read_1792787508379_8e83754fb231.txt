    88→interface ScenarioFieldProps {
    89→  id: string;

    91→  icon: ComponentType<{ className?: string }>;
    92→  children: ReactNode;

    95→
    96→function ScenarioField({ id, label, icon: Icon, children, help }: ScenarioFieldProps) {
    97→  return (
    98→    <div className="space-y-1.5">
    99→      <Label htmlFor={id} className="flex items-center gap-1.5 text-xs font-medium">
   100→        <Icon className="size-3.5 text-muted-foreground" />
   101→        {label}
   102→      </Label>
   103→      {children}
   104→      {help && <p className="text-[11px] leading-snug text-muted-foreground">{help}</p>}
   105→    </div>
   106→  );
   107→}
   108→
   109→function ScenarioNumberInput({
   110→  id,
   111→  value,
   112→  min,
   113→  max,
   114→  step,
   115→  onCommit,
   116→  ariaLabel,
   117→}: {
   118→  id: string;
   119→  value: number;
   120→  min?: number;
   121→  max?: number;
   122→  step?: number;
   123→  onCommit: (value: number) => void;
   124→  ariaLabel: string;
   125→}) {
   126→  const [draft, setDraft] = useState(String(value));
   127→  useEffect(() => {
   128→    setDraft(String(value));
   129→  }, [value]);
   130→  const commit = (raw: string) => {
   131→    const parsed = Number(raw.replace(",", "."));
   132→    if (!Number.isFinite(parsed)) {
   133→      setDraft(String(value));
   134→      return;
   135→    }
   136→    let next = parsed;
   137→    if (min !== undefined) next = Math.max(min, next);
   138→    if (max !== undefined) next = Math.min(max, next);
   139→    onCommit(next);
   140→  };
   141→  return (
   142→    <Input
   143→      id={id}
   144→      type="number"
   145→      inputMode="decimal"
   146→      value={draft}
   147→      min={min}
   148→      max={max}
   149→      step={step}
   150→      aria-label={ariaLabel}
   151→      onChange={(event) => setDraft(event.target.value)}
   152→      onBlur={() => commit(draft)}
   153→      onKeyDown={(event) => {
   154→        if (event.key === "Enter") commit(draft);
   155→      }}
   156→      className="h-8 tabular-nums"
   157→    />
   158→  );
   159→}
