   104→function ScenarioNumberInput({
   105→  id,
   106→  value,
   107→  min,
   108→  max,
   109→  step,
   110→  onCommit,
   111→  ariaLabel,
   112→}: {
   113→  id: string;
   114→  value: number;
   115→  min?: number;
   116→  max?: number;
   117→  step?: number;
   118→  onCommit: (value: number) => void;
   119→  ariaLabel: string;
   120→}) {
   121→  const [draft, setDraft] = useState(String(value));
   122→  useEffect(() => {
   123→    setDraft(String(value));
   124→  }, [value]);
   125→  const commit = (raw: string) => {
   126→    const parsed = Number(raw.replace(",", "."));
   127→    if (!Number.isFinite(parsed)) {
   128→      setDraft(String(value));
   129→      return;
   130→    }
   131→    let next = parsed;
   132→    if (min !== undefined) next = Math.max(min, next);
   133→    if (max !== undefined) next = Math.min(max, next);
   134→    onCommit(next);
   135→  };
   136→  return (
   137→    <Input
   138→      id={id}
   139→      type="number"
   140→      inputMode="decimal"
   141→      value={draft}
   142→      min={min}
   143→      max={max}
   144→      step={step}
   145→      aria-label={ariaLabel}
   146→      onChange={(event) => setDraft(event.target.value)}
   147→      onBlur={() => commit(draft)}
   148→      onKeyDown={(event) => {
   149→        if (event.key === "Enter") commit(draft);
   150→      }}
   151→      className="h-8 tabular-nums"
   152→    />
   153→  );
   154→}
