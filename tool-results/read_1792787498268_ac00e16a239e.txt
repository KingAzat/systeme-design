     1→/**
     2→ * SystemLab — diagnostics pédagogiques construits après la simulation.
     3→ *
     4→ * Règles (≥ 12) : saturation critique/avertissement, file instable, SPOF,
     5→ * cache inefficace, épuisement de connexions, cascade de timeouts, composant
     6→ * isolé, rejets de rate limiter, circuit breaker ouvert, lag de file, goulot
     7→ * principal, disponibilité insuffisante, cycles/boucles, clampages,
     8→ * composants visuels neutres.
     9→ *
    10→ * Les solutions sont adaptées au rôle ET au mix lecture/écriture observé :
    11→ * on ne recommande JAMAIS des réplicas de lecture pour un flux d'écritures
    12→ * dominant (les réplicas ne rejouent que les lectures).
    13→ */

    49→const SEVERITY_RANK: Record<DiagnosticSeverity, number> = {
    50→  critical: 0,

    54→
    55→interface Ctx {
    56→  input: DiagnosticsInput;
    57→  byId: Map<string, { rt: NodeRuntime; nm: NodeMetrics; ex: NodeExtra }>;
    58→}
    59→
    60→function pct(n: number, digits = 1): string {
    61→  return formatPercent(n, digits);
    62→}
    63→
    64→function mixReadShare(c: Ctx, nodeId: string): number {
    65→  const ex = c.input.extras.get(nodeId);
    66→  if (!ex) return c.input.scenario.readRatio;
    67→  const total = ex.readArrivals + ex.writeArrivals;
    68→  if (total <= 0) return c.input.scenario.readRatio;
    69→  return ex.readArrivals / total;
    70→}
    71→
    72→function sol(label: string, how: string, conditions: string, tradeoffs: string): DiagnosticSolution {
    73→  return { label, how, conditions, tradeoffs };
    74→}

    81→  const readsDominant = mix >= 0.5;
    82→  const out: DiagnosticSolution[] = [];
    83→  switch (rt.role) {
    84→    case "server":
    85→      out.push(
    86→        sol(
    87→          "Ajouter des instances",

    89→          "Le temps de traitement moyen est déjà raisonnable et le composant est stateless.",
    90→          "Coût proportionnel ; la charge aval (base, cache) augmente d'autant — vérifiez qu'elle suit.",
    91→        ),
    92→        sol(
    93→          "Optimiser le temps de traitement",
    94→          "Profiler le code (requêtes N+1, sérialisation, CPU) pour réduire avgServiceMs : c'est le seul levier qui améliore latence ET débit simultanément.",
    95→          "Le profilage révèle des goulots applicatifs (CPU, I/O, allocations).",
    96→          "Effort d'ingénierie ; gains plafonnés par les dépendances aval.",
    97→        ),
    98→      );
    99→      if (readsDominant) {
   100→        out.push(
   101→          sol(
   102→            "Mettre en cache les lectures",
   103→            "Intercaler un cache (Redis) entre les serveurs et la base pour les lectures répétées — le flux observé est à " +
   104→              pct(mix) +
   105→              " de lectures.",
   106→            "Forte réutilisation des mêmes lectures (sessions, catalogue, référence).",
   107→            "Incohérences possibles (TTL/invalidation) ; mémoire supplémentaire ; n'accélère pas les écritures.",
   108→          ),
   109→        );
   110→      } else {
   111→        out.push(
   112→          sol(
   113→            "Extraire les écritures vers une file",
   114→            "Publier les écritures dans une file et les traiter par des workers asynchrones : le serveur répond 202 immédiatement — le flux observé est à " +
   115→              pct(1 - mix) +
   116→              " d'écritures.",
   117→            "Les écritures n'exigent pas une réponse synchrone à l'utilisateur.",
   118→            "Cohérence à terme (eventual consistency) ; complexité opérationnelle ; supervision des files et retries.",
   119→          ),
   120→        );
   121→      }
   122→      break;
   123→    case "database": {
   124→      out.push(
   125→        sol(
   126→          "Augmenter le pool de connexions",
   127→          `Passer maxConnections de ${rt.rc.maxConnections} à ${Math.round(rt.rc.maxConnections * 1.5)} : capacité ≈ connexions / temps moyen.`,
   128→          "Le temps moyen par requête est sain et l'attente vient du pool.",
   129→          "Chaque connexion consomme mémoire + un process côté SGBD ; au-delà d'un point, le CPU et les verrous deviennent limitants.",
   130→        ),
   131→        sol(
   132→          "Réduire le temps par requête",
   133→          "Optimiser les requêtes (index, plans d'exécution, requêtes N+1) : agit sur avgReadMs/avgWriteMs, donc directement sur la capacité.",
   134→          "Le pool n'est pas saturé mais chaque requête est lente.",
   135→          "Nécessite du profilage (EXPLAIN, slow query log) ; maintenance des index.",
   136→        ),
   137→      );
   138→      if (readsDominant) {
   139→        out.push(
   140→          sol(
   141→            "Décharger les lectures (cache + réplicas)",
   142→            "Ajouter un cache applicatif pour les lectures chaudes et, au-delà, des réplicas de lecture — le flux arrivant est à " +
   143→              pct(mix) +
   144→              " de lectures, donc dimensionnable horizontalement.",
   145→            "Flux dominant en lectures avec réutilisation (la clé : le taux de hit).",
   146→            "Délai de réplication (lectures légèrement périmées) ; coût par réplica ; l'application doit router lectures/écritures.",
   147→          ),
   148→        );
   149→      } else {
   150→        out.push(
   151→          sol(
   152→            "Lisser et répartir les écritures",
   153→            "Le flux est à " +
   154→              pct(1 - mix) +
   155→              " d'écritures : ajouter des réplicas de lecture n'y changerait rien. lisser par une file tampon, réduire fsync par batch, partitionner/sharder les tables chaudes.",
   156→            "Écritures dominantes et besoin de tenir le débit.",
   157→            "Sharding = complexité applicative forte ; batching retarde la durabilité ; la file introduit de l'asynchrone.",
   158→          ),
   159→        );
   160→      }
   161→      break;
   162→    }
   163→    case "replica":

   178→          "Le taux de hit est limité par les évictions (clés expulsées avant leur réutilisation).",

   180→        ),
   181→        sol(
   182→          "Augmenter les connexions",
   183→          `maxConnections ${rt.rc.maxConnections} → ${Math.round(rt.rc.maxConnections * 1.5)} si l'attente vient du pool de connexions.`,

   187→      );
   188→      break;
   189→    case "balancer":
   190→    case "cdn":
   191→    case "proxy":
   192→    case "storage":
   193→      out.push(
   194→        sol(
   195→          "Augmenter la capacité",
   196→          `capacityRps ${formatRps(rt.rc.capacityRps)} → ${formatRps(rt.rc.capacityRps * 2)}${
   197→            rt.role === "balancer" ? " (ou le nombre d'instances du répartiteur)" : ""
   198→          }.`,
   199→          "Le rejet vient du dépassement de capacité en débit, pas des cibles aval.",
   200→          "Coût du service géré ; vérifier que l'aval absorbe le débit supplémentaire.",
   201→        ),
   202→      );
   203→      break;
   204→    case "queue":
   205→      out.push(
   206→        sol(
   207→          "Renforcer les workers aval",
   208→          "Ajouter des consommateurs : le débit d'une file est celui de ses workers (instances × concurrence / temps de traitement).",
   209→          "La profondeur de file grandit alors que les workers sont saturés.",
   210→          "La dépendance aval (base…) doit absorber le débit des workers supplémentaires.",
   211→        ),
   212→        sol(
   213→          "Ajouter des partitions",
   214→          `partitions ${rt.rc.partitions} → ${rt.rc.partitions * 2} : répartit le stockage et le débit d'ingestion.`,
   215→          "Le producteur est bridé par l'ingestion de la file elle-même.",
   216→          "Rééquilibrage des consommateurs ; clés de partition à choisir soigneusement.",
   217→        ),
   218→      );
   219→      break;
   220→    case "worker":
   221→      out.push(
   222→        sol(
   223→          "Ajouter des instances de worker",
   224→          `instances ${rt.rc.instances} → ${rt.rc.instances * 2} : le débit de traitement croît linéairement.`,
   225→          "Les messages s'accumulent en amont et les workers sont saturés.",
   226→          "Coût proportionnel ; la dépendance aval (base) reçoit le double de débit.",
   227→        ),
   228→        sol(
   229→          "Réduire avgProcessingMs",
   230→          "Optimiser le traitement (batch, I/O, algorithmique) : temps par message plus court ⇒ débit plus grand sans instance.",
   231→          "Le temps de traitement contient des phases optimisables (I/O séquentielles, batches).",
   232→          "Effort de développement ; certains traitements ne sont pas compressibles.",
   233→        ),
   234→      );
   235→      break;
   236→    default:
   237→      out.push(
   238→        sol(
   239→          "Renforcer ce composant",
   240→          "Augmenter sa capacité nominale ou le nombre d'instances.",
   241→          "Le composant est le facteur limitant du chemin.",
   242→          "Coût ; vérifier l'aval avant d'augmenter le débit amont.",
   243→        ),
   244→      );
   245→  }
   246→  return out;
   247→}

   249→/* ============================================================

   252→
   253→type Rule = (c: Ctx) => Diagnostic[];
   254→
   255→function mk(
   256→  id: string,
   257→  severity: DiagnosticSeverity,
   258→  title: string,
   259→  nodeId: string | undefined,
   260→  nodeLabel: string | undefined,
   261→  evidence: { label: string; value: string }[],
   262→  cause: string,
   263→  explanation: string,
   264→  solutions: DiagnosticSolution[],
   265→): Diagnostic {
   266→  // Identifiant garanti unique : suffixé par le nœud concerné le cas échéant.
   267→  const d: Diagnostic = {
   268→    id: nodeId ? `${id}--${nodeId}` : id,
   269→    severity,
   270→    title,
   271→    evidence,
   272→    cause,
   273→    explanation,
   274→    solutions,
   275→  };
   276→  if (nodeId) d.nodeId = nodeId;
   277→  if (nodeLabel) d.nodeLabel = nodeLabel;
   278→  return d;
   279→}
   280→
   281→/** R1/R2 — saturation (≥ 0,95 critique, ≥ 0,8 avertissement). */
   282→const ruleSaturation: Rule = (c) => {
   283→  const out: Diagnostic[] = [];
   284→  const entries = [...c.byId.values()]
   285→    .filter((e) => e.rt.role !== "generator" && e.nm.arrivals > 0)
   286→    .sort((a, b) => b.nm.utilization - a.nm.utilization || a.rt.id.localeCompare(b.rt.id));
   287→  for (const e of entries) {
   288→    const util = e.nm.utilization;
   289→    if (util < 0.8) continue;
   290→    const severity: DiagnosticSeverity = util >= 0.95 ? "critical" : "warning";
   291→    const mix = mixReadShare(c, e.rt.id);
   292→    out.push(
   293→      mk(
   294→        util >= 0.95 ? "saturation-critique" : "saturation-elevee",
   295→        severity,
   296→        util >= 0.95 ? "Saturation critique" : "Saturation élevée",
   297→        e.rt.id,
   298→        e.rt.label,
   299→        [
   300→          { label: "Utilisation moyenne", value: pct(util) },
   301→          { label: "Utilisation maximale", value: pct(e.nm.utilizationMax) },
   302→          { label: "Capacité théorique", value: formatRps(e.nm.capacityRps) },
   303→          { label: "Débit traité", value: formatRps(e.nm.throughputRps) },
   304→          { label: "Attente p95", value: formatMs(e.nm.p95WaitMs) },
   305→        ],
   306→        `Le composant fonctionne à ${pct(util)} de sa capacité (${formatRps(
   307→          e.nm.throughputRps,
   308→        )} traités pour ${formatRps(e.nm.capacityRps)} théoriques) : chaque requête arrive à trouver les ressources occupées.`,
   309→        `La loi de Little (L = λ × W) impose un choix : au-delà de ~80 % d'utilisation, l'attente moyenne croît de façon hyperbolique — doubler la charge résiduelle double peu le débit mais démultiplie la latence. Un composant saturé n'est pas seulement lent : il rejette (files pleines) et propage l'attente à toute la chaîne amont, qui finit elle-même par saturer. Les solutions ci-dessous sont ordonnées par rapport efficacité/coût pour ce rôle et ce mix (${pct(
   310→          mix,
   311→        )} de lectures).`,
   312→        capacitySolutions(c, e.rt, mix),
   313→      ),
   314→    );
   315→  }
   316→  return out.slice(0, 4); // éviter le spam : les 4 plus saturés
   317→};
   318→
   319→/** R3 — file instable (ρ ≥ 1). */
   320→const ruleUnstableQueue: Rule = (c) => {
   321→  const out: Diagnostic[] = [];
   322→  for (const e of c.byId.values()) {
   323→    if (e.rt.role === "generator" || e.nm.arrivals <= 0) continue;
   324→    if (e.nm.utilization < 1 && e.rt.role !== "queue") continue;
   325→    if (e.rt.role === "queue") {
   326→      // files : instabilité = profondeur qui ne se vide pas
   327→      if (e.ex.queueDepthEnd <= 0 || e.nm.maxQueueLength <= 0) continue;
   328→      if (e.ex.queueDepthEnd < Math.max(5, e.ex.queueDepthMax * 0.25)) continue;
   329→    } else if (e.nm.utilization < 1) {
   330→      continue;
   331→    }
   332→    const mix = mixReadShare(c, e.rt.id);
   333→    out.push(
   334→      mk(
   335→        "file-instable",
   336→        "critical",
   337→        "File d'attente instable (ρ ≥ 1)",
   338→        e.rt.id,
   339→        e.rt.label,
   340→        [
   341→          { label: "Utilisation moyenne", value: pct(e.nm.utilization) },
   342→          { label: "File moyenne", value: `${formatNumber(Math.round(e.nm.avgQueueLength))} requêtes` },
   343→          { label: "File maximale", value: `${formatNumber(Math.round(e.nm.maxQueueLength))} requêtes` },
   344→          { label: "Débit traité", value: formatRps(e.nm.throughputRps) },
   345→        ],
   346→        `Le taux d'arrivée dépasse la capacité de service : la file s'allonge sans limite pendant la simulation (${formatNumber(
   347→          Math.round(e.nm.avgQueueLength),
   348→        )} requêtes en moyenne, ${formatNumber(Math.round(e.nm.maxQueueLength))} au pic).`,
   349→        "Une file est stable si et seulement si ρ = λ / µ < 1. À ρ ≥ 1, chaque seconde ajoute (λ − µ) requêtes : la latence croît linéairement avec le temps et le système ne rattrape jamais, même pendant les creux. Aucun buffering ne sauve un système sous-dimensionné — la file décale le problème (et la douleur) au lieu de le résoudre.",
   350→        capacitySolutions(c, e.rt, mix).concat([
   351→          sol(
   352→            "Limiter le débit entrant",
   353→            "Placer un rate limiter ou une file tampon bornée en amont : dégrader explicitement (429/503) vaut mieux que laisser la latence exploser pour tout le monde.",
   354→            "Le pic de charge est temporaire et le client sait réessayer.",
   355→            "Rejets visibles pour l'utilisateur ; à documenter dans le contrat de service (SLA).",
   356→          ),
   357→        ]),
   358→      ),
   359→    );
   360→  }
   361→  return out.slice(0, 3);
   362→};
   363→
   364→/** R4 — point de défaillance unique. */

   368→  for (const e of c.byId.values()) {
   369→    if (e.nm.arrivals <= 0) continue;
   370→    if (trafficRoles.has(e.rt.role) && e.rt.rc.instances <= 1) {
   371→      out.push(
   372→        mk(
   373→          "spof",
   374→          "warning",
   375→          "Point de défaillance unique (1 instance)",
   376→          e.rt.id,
   377→          e.rt.label,
   378→          [
   379→            { label: "Instances", value: "1" },
   380→            { label: "Rôle", value: e.rt.label },
   381→            { label: "Débit traité", value: formatRps(e.nm.throughputRps) },
   382→          ],
   383→          "Ce composant traite du trafic avec une seule instance : sa panne interrompt tout le chemin qui passe par lui.",
   384→          "La haute disponibilité se construit par redondance : N+1 instances derrière un répartiteur avec health checks. Une instance unique cumule les modes de défaillance (panne machine, déploiement, mise à jour) — le « zéro downtime » exige au moins deux copies actives.",
   385→          [
   386→            sol(
   387→              "Passer à 2 instances",
   388→              "Dupliquer le composant et répartir la charge (round-robin ou least-connections) entre les copies.",
   389→              "Le composant est stateless ou son état est partagé (base, cache).",
   390→              "Coût ×2 ; l'état local (sessions en mémoire) doit être externalisé.",
   391→            ),
   392→            sol(
   393→              "Activer les health checks",
   394→              "Configurer le répartiteur amont pour écarter automatiquement les cibles en panne et basculer le trafic.",
   395→              "Il existe déjà un répartiteur en amont.",
   396→              "Un délai de détection (intervalle des sondes) pendant lequel les requêtes échouent quand même.",
   397→            ),
   398→          ],
   399→        ),
   400→      );
   401→    }
   402→    // base primaire sans réplica connecté
   403→    if (e.rt.role === "database") {
   404→      const hasReplica = c.input.runtimes.some((r) => r.role === "replica");
   405→      if (!hasReplica) {
   406→        out.push(
   407→          mk(
   408→            "spof-base",
   409→            "warning",
   410→            "Base primaire sans redondance",
   411→            e.rt.id,
   412→            e.rt.label,
   413→            [
   414→              { label: "Réplicas de lecture", value: "0" },
   415→              { label: "Politique de réplication", value: e.rt.rc.replicationPolicy },
   416→            ],
   417→            "La base de données est la source de vérité et n'a aucun réplica/standby simulé : une panne y est définitive (perte de service, restauration longue).",
   418→            "Une base primaire unique concentre à la fois le flux d'écritures (non dimensionnable par réplicas) et le risque de disponibilité. Un standby (même asynchrone) permet un failover en minutes ; les sauvegardes PITR bornent la perte de données. À ne pas confondre : un standby de disponibilité n'accélère pas les écritures.",
   419→            [
   420→              sol(
   421→                "Ajouter un standby de failover",
   422→                "Répliquer la base (streaming asynchrone) vers un nœud standby prêt à être promu.",
   423→                "Le SLO de disponibilité exige un RTO de l'ordre de la minute.",
   424→                "Coût d'un nœud complet ; le failover asynchrone peut perdre les dernières transactions (RPO > 0).",
   425→              ),
   426→              sol(
   427→                "Sauvegardes + PITR",
   428→                "Sauvegardes régulières + WAL archivé pour restaurer à un instant donné.",
   429→                "Toute architecture, avec ou sans standby.",
   430→                "RTO en heures ; à tester (une sauvegarde non testée n'existe pas).",
   431→              ),
   432→            ],
   433→          ),
   434→        );
   435→      }
   436→    }
   437→  }
   438→  return out.slice(0, 4);
   439→};

   441→/** R5 — cache inefficace. */
   442→const ruleCacheHit: Rule = (c) => {
   443→  const out: Diagnostic[] = [];
   444→  for (const e of c.byId.values()) {
   445→    if ((e.rt.role !== "cache" && e.rt.role !== "cdn") || e.nm.arrivals < 20) continue;
   446→    const hitRate = e.nm.cacheHitRate;
   447→    if (hitRate === undefined || hitRate >= 0.5) continue;
   448→    out.push(
   449→      mk(
   450→        "cache-hit-faible",
   451→        "warning",
   452→        "Cache inefficace (taux de hit faible)",
   453→        e.rt.id,
   454→        e.rt.label,
   455→        [
   456→          { label: "Taux de hit", value: pct(hitRate) },
   457→          { label: "Requêtes servies du cache", value: formatNumber(Math.round(e.nm.cacheHits ?? 0)) },
   458→          { label: "Lectures reçues", value: formatNumber(Math.round(e.ex.readArrivals * (c.input.scale > 0 ? 1 / c.input.scale : 1))) },
   459→        ],
   460→        `Seules ${pct(hitRate)} des lectures sont servies par le cache : la majorité du trafic traverse quand même vers l'origine.`,

  1027→const MAX_DIAGNOSTICS = 12;
  1028→
  1029→export function buildDiagnostics(input: DiagnosticsInput): Diagnostic[] {
  1030→  const byId = new Map<string, { rt: NodeRuntime; nm: NodeMetrics; ex: NodeExtra }>();
  1031→  for (const rt of input.runtimes) {
  1032→    const nm = input.nodes.find((n) => n.nodeId === rt.id);
  1033→    if (!nm) continue;
  1034→    const ex =
  1035→      input.extras.get(rt.id) ??
  1036→      ({
  1037→        nodeId: rt.id,
  1038→        downstreamP95Ms: 0,
  1039→        cbOpenCount: 0,
  1040→        cbLastOpenAtSec: null,
  1041→        connExhaustedErrors: 0,
  1042→        readArrivals: 0,
  1043→        writeArrivals: 0,
  1044→        queueDepthEnd: 0,
  1045→        queueDepthMax: 0,
  1046→      } as NodeExtra);
  1047→    byId.set(rt.id, { rt, nm, ex });
  1048→  }
  1049→  const c: Ctx = { input, byId };
  1050→
  1051→  const rules: Rule[] = [
  1052→    ruleSaturation,
  1053→    ruleUnstableQueue,
  1054→    ruleConnExhausted,
  1055→    ruleTimeoutCascade,
  1056→    ruleAvailability,
  1057→    ruleRateLimit,
  1058→    ruleSpof,
  1059→    ruleCacheHit,
  1060→    ruleCircuitOpen,
  1061→    ruleQueueLag,
  1062→    ruleBottleneck,
  1063→    ruleLatency,
  1064→    ruleIsolated,
  1065→    ruleCycles,
  1066→    ruleVisual,
  1067→    ruleClamps,
  1068→  ];
  1069→
  1070→  const all: Diagnostic[] = [];
  1071→  for (const rule of rules) {
  1072→    all.push(...rule(c));
  1073→  }
  1074→  // tri par gravité, ordre de règle stable à gravité égale
  1075→  all.sort((a, b) => SEVERITY_RANK[a.severity] - SEVERITY_RANK[b.severity]);
  1076→  return all.slice(0, MAX_DIAGNOSTICS);
  1077→}
