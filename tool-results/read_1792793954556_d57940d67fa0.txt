   146→/** Saisie numérique avec ± ; commit sur blur/Entrée (les +/- commitent direct). */
   147→function NumberInput({ id, value, min, max, step, ariaLabel, onCommit }: NumberInputProps) {
   148→  const [draft, setDraft] = useState<string>(String(value));
   149→  useEffect(() => {
   150→    setDraft(String(value));
   151→  }, [value]);
   152→
   153→  const commit = (raw: string) => {
   154→    const parsed = Number(raw.replace(",", "."));
   155→    if (!Number.isFinite(parsed)) {
   156→      setDraft(String(value));
   157→      return;
   158→    }
   159→    let next = parsed;
   160→    if (min !== undefined) next = Math.max(min, next);
   161→    if (max !== undefined) next = Math.min(max, next);
   162→    onCommit(roundTo(next));
   163→  };
   164→
   165→  const stepBy = (direction: 1 | -1) => {
   166→    const s = step ?? 1;
   167→    const current = Number(draft.replace(",", ".")) || 0;
   168→    let next = roundTo(current + direction * s);
   169→    if (min !== undefined) next = Math.max(min, next);
   170→    if (max !== undefined) next = Math.min(max, next);
   171→    setDraft(String(next));
   172→    onCommit(next);
   173→  };
   174→
   175→  return (
   176→    <div className="flex items-center gap-1">
   177→      <Button
   178→        type="button"
   179→        variant="outline"
   180→        size="icon"
   181→        className="size-7 shrink-0"
   182→        aria-label={`Diminuer ${ariaLabel}`}
   183→        onClick={() => stepBy(-1)}
   184→      >
   185→        <Minus className="size-3" />
   186→      </Button>
   187→      <Input
   188→        id={id}
   189→        type="number"
   190→        inputMode="decimal"
   191→        value={draft}
   192→        min={min}
   193→        max={max}
   194→        step={step}
   195→        aria-label={ariaLabel}
   196→        onChange={(event) => setDraft(event.target.value)}
   197→        onBlur={() => commit(draft)}
   198→        onKeyDown={(event) => {
   199→          if (event.key === "Enter") commit(draft);
   200→        }}
   201→        className="h-8 text-center text-sm tabular-nums [appearance:textfield] [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
   202→      />
   203→      <Button
   204→        type="button"
   205→        variant="outline"
   206→        size="icon"
   207→        className="size-7 shrink-0"
   208→        aria-label={`Augmenter ${ariaLabel}`}
   209→        onClick={() => stepBy(1)}
   210→      >
   211→        <Plus className="size-3" />
   212→      </Button>
   213→    </div>
   214→  );
   215→}

   221→function ConfigField({ node, field }: { node: ArchNode; field: CatalogField }) {
   222→  const updateNodeConfig = useEditorStore((s) => s.updateNodeConfig);
   223→  const value = node.config[field.key];
   224→  const fieldId = `node-${node.id}-${field.key}`;
   225→
   226→  const commit = (next: number | string | boolean) => {
   227→    useEditorStore.getState().updateNodeConfig(node.id, field.key, next);
   228→  };
   229→
   230→  if (field.type === "boolean") {
   231→    const checked = Boolean(value);
   232→    return (
   233→      <div className="flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-3 py-2.5">
   234→        <LabelWithHelp field={field} />
   235→        <Switch
   236→          id={fieldId}
   237→          checked={checked}
   238→          onCheckedChange={commit}
   239→          aria-label={field.label}
   240→        />
   241→      </div>
   242→    );
   243→  }
   244→
   245→  if (field.type === "select") {
   246→    return (
   247→      <div className="space-y-1.5">
   248→        <LabelWithHelp field={field} />
   249→        <Select
   250→          value={typeof value === "string" ? value : String(value)}
   251→          onValueChange={(next) => commit(next)}
   252→        >
   253→          <SelectTrigger id={fieldId} className="h-8 w-full" aria-label={field.label}>
   254→            <SelectValue placeholder={field.label} />
   255→          </SelectTrigger>
   256→          <SelectContent>
   257→            {field.options?.map((option) => (
   258→              <SelectItem key={option.value} value={option.value}>
   259→                {option.label}
   260→              </SelectItem>
   261→            ))}
   262→          </SelectContent>
   263→        </Select>
   264→      </div>
   265→    );
   266→  }
   267→
   268→  if (field.type === "text") {
   269→    return (
   270→      <div className="space-y-1.5">
   271→        <LabelWithHelp field={field} />
   272→        <Input
   273→          id={fieldId}
   274→          value={typeof value === "string" ? value : String(value ?? "")}
   275→          onChange={(event) => commit(event.target.value)}
   276→          aria-label={field.label}
   277→          className="h-8"
   278→        />
   279→      </div>
   280→    );
   281→  }
   282→
   283→  // type === "number"
   284→  const numeric = typeof value === "number" ? value : Number(value) || 0;
   285→  const hasBounds = field.min !== undefined && field.max !== undefined;
   286→  return (
   287→    <div className="space-y-1.5">
   288→      <LabelWithHelp field={field} />
   289→      <NumberInput
   290→        id={fieldId}
   291→        value={numeric}
   292→        min={field.min}
   293→        max={field.max}
   294→        step={field.step}
   295→        ariaLabel={field.label}
   296→        onCommit={commit}
   297→      />
   298→      {hasBounds && (
   299→        <div className="flex items-center gap-3 pt-0.5">
   300→          <Slider
   301→            min={field.min}
   302→            max={field.max}
   303→            step={field.step ?? 1}
   304→            value={[numeric]}
   305→            onValueChange={(values) =>
   306→              useEditorStore
   307→                .getState()
   308→                .updateNodeConfig(node.id, field.key, values[0])
   309→            }
   310→            onValueCommit={(values) => commit(values[0])}
   311→            aria-label={field.label}
   312→            className="flex-1"
   313→          />
   314→          <span className="w-14 shrink-0 text-right text-xs text-muted-foreground tabular-nums">
   315→            {formatNumber(numeric)}
   316→          </span>
   317→        </div>
   318→      )}
   319→    </div>
   320→  );
   321→}

   384→/** Estimations analytiques, débouncées 300 ms et mémoïsées par graphe. */
   385→function useNodeEstimate(nodeId: string): NodeEstimate | null | "pending" {
   386→  const graphKey = useEditorStore((s) => JSON.stringify([s.nodes, s.edges, s.scenario]));
   387→  const cache = useRef(new Map<string, NodeEstimate[]>());
   388→  const [state, setState] = useState<{ estimate: NodeEstimate | null; ready: boolean }>({
   389→    estimate: null,
   390→    ready: false,
   391→  });
   392→
   393→  useEffect(() => {
   394→    const timer = window.setTimeout(() => {
   395→      const cached = cache.current.get(graphKey);
   396→      if (cached) {
   397→        setState({ estimate: cached.find((e) => e.nodeId === nodeId) ?? null, ready: true });
   398→        return;
   399→      }
   400→      const doc = useEditorStore.getState();
   401→      const computed = estimateArchitecture({
   402→        name: doc.name,
   403→        nodes: doc.nodes,
   404→        edges: doc.edges,
   405→        zones: doc.zones,
   406→        scenario: doc.scenario,
   407→      });
   408→      if (cache.current.size > 24) cache.current.clear();
   409→      cache.current.set(graphKey, computed);
   410→      setState({ estimate: computed.find((e) => e.nodeId === nodeId) ?? null, ready: true });
   411→    }, 300);
   412→    return () => window.clearTimeout(timer);
   413→  }, [graphKey, nodeId]);
   414→
   415→  return state.ready ? state.estimate : "pending";
   416→}
