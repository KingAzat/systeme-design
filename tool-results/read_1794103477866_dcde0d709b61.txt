   142→/** Saisie numérique avec ± ; commit sur blur/Entrée (les +/- commitent direct). */
   143→function NumberInput({ id, value, min, max, step, ariaLabel, onCommit }: NumberInputProps) {
   144→  const [draft, setDraft] = useState<string>(String(value));
   145→  useEffect(() => {
   146→    setDraft(String(value));
   147→  }, [value]);
   148→
   149→  const commit = (raw: string) => {
   150→    const parsed = Number(raw.replace(",", "."));
   151→    if (!Number.isFinite(parsed)) {
   152→      setDraft(String(value));
   153→      return;
   154→    }
   155→    let next = parsed;
   156→    if (min !== undefined) next = Math.max(min, next);
   157→    if (max !== undefined) next = Math.min(max, next);
   158→    onCommit(roundTo(next));
   159→  };
   160→
   161→  const stepBy = (direction: 1 | -1) => {
   162→    const s = step ?? 1;
   163→    const current = Number(draft.replace(",", ".")) || 0;
   164→    let next = roundTo(current + direction * s);
   165→    if (min !== undefined) next = Math.max(min, next);
   166→    if (max !== undefined) next = Math.min(max, next);
   167→    setDraft(String(next));
   168→    onCommit(next);
   169→  };
   170→
   171→  return (
   172→    <div className="flex items-center gap-1">
   173→      <Button
   174→        type="button"
   175→        variant="outline"
   176→        size="icon"
   177→        className="size-7 shrink-0"
   178→        aria-label={`Diminuer ${ariaLabel}`}
   179→        onClick={() => stepBy(-1)}
   180→      >
   181→        <Minus className="size-3" />
   182→      </Button>
   183→      <Input
   184→        id={id}
   185→        type="number"
   186→        inputMode="decimal"
   187→        value={draft}
   188→        min={min}
   189→        max={max}
   190→        step={step}
   191→        aria-label={ariaLabel}
   192→        onChange={(event) => setDraft(event.target.value)}
   193→        onBlur={() => commit(draft)}
   194→        onKeyDown={(event) => {
   195→          if (event.key === "Enter") commit(draft);
   196→        }}
   197→        className="h-8 text-center text-sm tabular-nums [appearance:textfield] [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
   198→      />
   199→      <Button
   200→        type="button"
   201→        variant="outline"
   202→        size="icon"
   203→        className="size-7 shrink-0"
   204→        aria-label={`Augmenter ${ariaLabel}`}
   205→        onClick={() => stepBy(1)}
   206→      >
   207→        <Plus className="size-3" />
   208→      </Button>
   209→    </div>
   210→  );
   211→}

   217→function ConfigField({ node, field }: { node: ArchNode; field: CatalogField }) {
   218→  const updateNodeConfig = useEditorStore((s) => s.updateNodeConfig);
   219→  const value = node.config[field.key];
   220→  const fieldId = `node-${node.id}-${field.key}`;
   221→
   222→  const commit = (next: number | string | boolean) => {
   223→    useEditorStore.getState().updateNodeConfig(node.id, field.key, next);
   224→  };
   225→
   226→  if (field.type === "boolean") {
   227→    const checked = Boolean(value);
   228→    return (
   229→      <div className="flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-3 py-2.5">
   230→        <LabelWithHelp field={field} />
   231→        <Switch
   232→          id={fieldId}
   233→          checked={checked}
   234→          onCheckedChange={commit}
   235→          aria-label={field.label}
   236→        />
   237→      </div>
   238→    );
   239→  }
   240→
   241→  if (field.type === "select") {
   242→    return (
   243→      <div className="space-y-1.5">
   244→        <LabelWithHelp field={field} />
   245→        <Select
   246→          value={typeof value === "string" ? value : String(value)}
   247→          onValueChange={(next) => commit(next)}
   248→        >
   249→          <SelectTrigger id={fieldId} className="h-8 w-full" aria-label={field.label}>
   250→            <SelectValue placeholder={field.label} />
   251→          </SelectTrigger>
   252→          <SelectContent>
   253→            {field.options?.map((option) => (
   254→              <SelectItem key={option.value} value={option.value}>
   255→                {option.label}
   256→              </SelectItem>
   257→            ))}
   258→          </SelectContent>
   259→        </Select>
   260→      </div>
   261→    );
   262→  }
   263→
   264→  if (field.type === "text") {
   265→    return (
   266→      <div className="space-y-1.5">
   267→        <LabelWithHelp field={field} />
   268→        <Input
   269→          id={fieldId}
   270→          value={typeof value === "string" ? value : String(value ?? "")}
   271→          onChange={(event) => commit(event.target.value)}
   272→          aria-label={field.label}
   273→          className="h-8"
   274→        />
   275→      </div>
   276→    );
   277→  }
   278→
   279→  // type === "number"
   280→  const numeric = typeof value === "number" ? value : Number(value) || 0;
   281→  const hasBounds = field.min !== undefined && field.max !== undefined;
   282→  return (
   283→    <div className="space-y-1.5">
   284→      <LabelWithHelp field={field} />
   285→      <NumberInput
   286→        id={fieldId}
   287→        value={numeric}
   288→        min={field.min}
   289→        max={field.max}
   290→        step={field.step}
   291→        ariaLabel={field.label}
   292→        onCommit={commit}
   293→      />
   294→      {hasBounds && (
   295→        <div className="flex items-center gap-3 pt-0.5">
   296→          <Slider
   297→            min={field.min}
   298→            max={field.max}
   299→            step={field.step ?? 1}
   300→            value={[numeric]}
   301→            onValueChange={(values) =>
   302→              useEditorStore
   303→                .getState()
   304→                .updateNodeConfig(node.id, field.key, values[0])
   305→            }
   306→            onValueCommit={(values) => commit(values[0])}
   307→            aria-label={field.label}
   308→            className="flex-1"
   309→          />
   310→          <span className="w-14 shrink-0 text-right text-xs text-muted-foreground tabular-nums">
   311→            {formatNumber(numeric)}
   312→          </span>
   313→        </div>
   314→      )}
   315→    </div>
   316→  );
   317→}

   380→/** Estimations analytiques, débouncées 300 ms et mémoïsées par révision. */
   381→function useNodeEstimate(nodeId: string): NodeEstimate | null | "pending" {
   382→  // `revision` est incrémenté par toute mutation de CONTENU (nodes, edges,
   383→  // zones, scenario, name) — jamais par la sélection — et setScenario crée un
   384→  // nouvel objet scenario tout en incrémentant revision : revision seule
   385→  // couvre donc le recalcul ET la clé de cache. Le sélecteur retourne un
   386→  // nombre (coût O(1) à chaque set() du store) à la place de l'ancien
   387→  // JSON.stringify du graphe entier, qui s'exécutait à chaque frame de drag.
   388→  const revision = useEditorStore((s) => s.revision);
   389→  const cache = useRef(new Map<number, NodeEstimate[]>());
   390→  const [state, setState] = useState<{ estimate: NodeEstimate | null; ready: boolean }>({
   391→    estimate: null,
   392→    ready: false,
   393→  });
   394→
   395→  useEffect(() => {
   396→    const timer = window.setTimeout(() => {
   397→      const cached = cache.current.get(revision);
   398→      if (cached) {
   399→        setState({ estimate: cached.find((e) => e.nodeId === nodeId) ?? null, ready: true });
   400→        return;
   401→      }
   402→      const doc = useEditorStore.getState();
   403→      const computed = estimateArchitecture({
   404→        name: doc.name,
   405→        nodes: doc.nodes,
   406→        edges: doc.edges,
   407→        zones: doc.zones,
   408→        scenario: doc.scenario,
   409→      });
   410→      if (cache.current.size > 24) cache.current.clear();
   411→      cache.current.set(revision, computed);
   412→      setState({ estimate: computed.find((e) => e.nodeId === nodeId) ?? null, ready: true });
   413→    }, 300);
   414→    return () => window.clearTimeout(timer);
   415→  }, [revision, nodeId]);
   416→
   417→  return state.ready ? state.estimate : "pending";
   418→}
