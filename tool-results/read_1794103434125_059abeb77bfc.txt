    39→
    40→function node(
    41→  id: string,
    42→  componentId: string,
    43→  label: string,
    44→  x: number,
    45→  y: number,
    46→  config: Record<string, ConfigValue> = {},
    47→  zoneId?: string,
    48→): ArchNode {
    49→  const base = COMPONENT_MAP[componentId]?.defaults ?? {};
    50→  return {
    51→    id,
    52→    componentId,
    53→    label,
    54→    config: { ...base, ...config },
    55→    position: { x, y },
    56→    zoneId,
    57→  };
    58→}
    59→
    60→function edge(
    61→  id: string,
    62→  source: string,
    63→  target: string,
    64→  protocol: Protocol = "http",
    65→  networkLatencyMs = 2,
    66→  bandwidthMbps = 100,
    67→  label?: string,
    68→): ArchEdge {
    69→  return { id, source, target, protocol, networkLatencyMs, bandwidthMbps, label };
    70→}
    71→
    72→function zone(
    73→  id: string,
    74→  label: string,
    75→  color: ZoneColor,
    76→  x: number,
    77→  y: number,
    78→  width: number,
    79→  height: number,
    80→): ArchZone {
    81→  return { id, label, color, position: { x, y }, size: { width, height } };
    82→}
    83→
    84→function data(
    85→  name: string,
    86→  description: string,
    87→  nodes: ArchNode[],
    88→  edges: ArchEdge[],
    89→  zones: ArchZone[],
    90→  scenarioConfig: ScenarioConfig,
    91→): ArchitectureData {
    92→  return { name, description, nodes, edges, zones, scenario: scenarioConfig };
    93→}
