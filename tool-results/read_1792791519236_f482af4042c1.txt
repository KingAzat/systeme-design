   181→   181→const HISTORY_CAPACITY = 50;
   182→   182→const COALESCE_WINDOW_MS = 800;
   183→   183→
   184→   184→/** Flag de session de drag (hors état React : pas de re-render attendu). */
   185→   185→let dragActive = false;
   186→   186→/** Dernière poussée d'historique coalescée (tag + horodatage). */
   187→   187→let lastCoalesced: { tag: string; at: number } | null = null;
   188→   188→
   189→   189→function snapshotOf(s: EditorState): DocSnapshot {
   190→   190→  // Sérialisation JSON : garantit l'absence de références partagées.
   191→   191→  return JSON.parse(
   192→   192→    JSON.stringify({
   193→   193→      name: s.name,
   194→   194→      description: s.description,
   195→   195→      nodes: s.nodes,
   196→   196→      edges: s.edges,
   197→   197→      zones: s.zones,
   198→   198→      scenario: s.scenario,
   199→   199→    }),
   200→   200→  ) as DocSnapshot;
   201→   201→}
   202→   202→
   203→   203→/** Partiel {past, future} à merger dans set() pour pousser l'historique. */
   204→   204→function historyPush(s: EditorState): { past: DocSnapshot[]; future: DocSnapshot[] } {
   205→   205→  return {
   206→   206→    past: [...s.past, snapshotOf(s)].slice(-HISTORY_CAPACITY),
   207→   207→    future: [],
   208→   208→  };
   209→   209→}
   210→   210→
   211→   211→/**
   212→   212→ * Coalescence : renvoie true s'il faut pousser un snapshot (début de rafale),
   213→   213→ * false si la modification fait suite à une poussée récente du même tag.
   214→   214→ */
   215→   215→function shouldPushCoalesced(tag: string, windowMs = COALESCE_WINDOW_MS): boolean {
   216→   216→  const now = Date.now();
   217→   217→  if (lastCoalesced && lastCoalesced.tag === tag && now - lastCoalesced.at < windowMs) {
   218→   218→    lastCoalesced.at = now; // prolonge la session d'édition continue
   219→   219→    return false;
   220→   220→  }
   221→   221→  lastCoalesced = { tag, at: now };
   222→   222→  return true;
   223→   223→}

   225→   225→function applySnapshot(snap: DocSnapshot): {
   226→   226→  name: string;
   227→   227→  description: string;
   228→   228→  nodes: ArchNode[];
   229→   229→  edges: ArchEdge[];
   230→   230→  zones: ArchZone[];
   231→   231→  scenario: ScenarioConfig;
   232→   232→} {
   233→   233→  return {
   234→   234→    name: snap.name,
   235→   235→    description: snap.description,
   236→   236→    nodes: snap.nodes.map((n) => ({ ...n, config: { ...n.config } })),
   237→   237→    edges: snap.edges.map((e) => ({ ...e })),
   238→   238→    zones: snap.zones.map((z) => ({ ...z })),
   239→   239→    scenario: { ...snap.scenario },
   240→   240→  };
   241→   241→}

   243→   243→function deepCloneNodes(nodes: ArchNode[]): ArchNode[] {
   244→   244→  return nodes.map((n) => ({ ...n, config: { ...n.config }, position: { ...n.position } }));
   245→   245→}
   246→   246→
   247→   247→function deepCloneEdges(edges: ArchEdge[]): ArchEdge[] {
   248→   248→  return edges.map((e) => ({ ...e }));
   249→   249→}
   250→   250→
   251→   251→function deepCloneZones(zones: ArchZone[]): ArchZone[] {
   252→   252→  return zones.map((z) => ({
   253→   253→    ...z,
   254→   254→    position: { ...z.position },
   255→   255→    size: { ...z.size },
   256→   256→  }));
   257→   257→}

   259→   259→/** Le nœud est-il dans le rectangle de la zone ? */
   260→   260→function pointInZone(
   261→   261→  position: { x: number; y: number },
   262→   262→  zone: ArchZone,
   263→   263→): boolean {
   264→   264→  return (
   265→   265→    position.x >= zone.position.x &&
   266→   266→    position.x <= zone.position.x + zone.size.width &&
   267→   267→    position.y >= zone.position.y &&
   268→   268→    position.y <= zone.position.y + zone.size.height
   269→   269→  );
   270→   270→}

   272→   272→/** Protocole par défaut d'une nouvelle connexion, déduit des rôles simulés. */
   273→   273→function defaultProtocol(source: ArchNode, target: ArchNode): Protocol {
   274→   274→  const roles = [
   275→   275→    getComponentById(source.componentId)?.simRole,
   276→   276→    getComponentById(target.componentId)?.simRole,
   277→   277→  ];
   278→   278→  if (roles.includes("queue")) return "amqp";
   279→   279→  if (roles.includes("cache")) return "redis";
   280→   280→  if (roles.includes("database") || roles.includes("replica")) return "sql";
   281→   281→  if (roles.includes("cdn")) return "https";
   282→   282→  return "http";
   283→   283→}

   694→   694→    const s = get();
   695→   695→    const removingNodes = s.selectedNodeIds;
   696→   696→    const removingEdge = s.selectedEdgeId;
   697→   697→    const removingZone = s.selectedZoneId;
   698→   698→    if (
   699→   699→      removingNodes.length === 0 &&
   700→   700→      !removingEdge &&
   701→   701→      !removingZone
   702→   702→    ) {
   703→   703→      return;
   704→   704→    }
   705→   705→    let nodes = s.nodes;
   706→   706→    let edges = s.edges;
   707→   707→    let zones = s.zones;
   708→   708→    if (removingNodes.length > 0) {
   709→   709→      const removed = new Set(removingNodes);
   710→   710→      nodes = s.nodes.filter((n) => !removed.has(n.id));
   711→   711→      edges = s.edges.filter((e) => !removed.has(e.source) && !removed.has(e.target));
   712→   712→    } else if (removingEdge) {
   713→   713→      edges = s.edges.filter((e) => e.id !== removingEdge);
   714→   714→    } else if (removingZone) {
   715→   715→      zones = s.zones.filter((z) => z.id !== removingZone);
   716→   716→      nodes = s.nodes.map((n) => (n.zoneId === removingZone ? { ...n, zoneId: undefined } : n));
   717→   717→    }
   718→   718→    // Une panne simulée visant un nœud supprimé est retirée du scénario.
   719→   719→    const failure = s.scenario.failure;
   720→   720→    const scenario =
   721→   721→      failure && removingNodes.includes(failure.nodeId)
   722→   722→        ? { ...s.scenario, failure: null }
   723→   723→        : s.scenario;
   724→   724→    set({
   725→   725→      ...historyPush(s),
   726→   726→      nodes,
   727→   727→      edges,
   728→   728→      zones,
   729→   729→      scenario,
   730→   730→      dirty: true,
   731→   731→      revision: s.revision + 1,
   732→   732→      ...EMPTY_SELECTION,
   733→   733→    });

   740→   740→    const clones: ArchNode[] = s.nodes
   741→   741→      .filter((n) => s.selectedNodeIds.includes(n.id))

   750→   750→        };
   751→   751→      });
   752→   752→    if (clones.length === 0) return;
   753→   753→    const clonedEdges: ArchEdge[] = s.edges
   754→   754→      .filter((e) => idMap.has(e.source) && idMap.has(e.target))
   755→   755→      .map((e) => ({

   763→   763→      nodes: [...s.nodes, ...clones],
   764→   764→      edges: [...s.edges, ...clonedEdges],
   765→   765→      dirty: true,
   766→   766→      revision: s.revision + 1,
   767→   767→      selectedNodeIds: clones.map((c) => c.id),
   768→   768→      selectedNodeId: clones[clones.length - 1].id,

   771→   771→    });
